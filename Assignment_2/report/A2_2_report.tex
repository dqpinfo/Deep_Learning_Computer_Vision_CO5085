\section{Chủ đề và bài toán}
\subsection{Bài toán}
\st{Phân đoạn ngữ nghĩa} là gán cho \emph{mỗi điểm ảnh} một nhãn lớp. Khác với phân loại ảnh ở A1 (một nhãn cho cả ảnh),
mô hình phải trả lời đồng thời ``ở đâu'' và ``là gì''. Nhóm định nghĩa bài toán trên Oxford-IIIT Pet như sau:
\begin{itemize}
  \item \st{Đầu vào:} một ảnh RGB chứa một con chó hoặc mèo.
  \item \st{Đầu ra:} bản đồ nhãn cùng kích thước, mỗi điểm ảnh thuộc một trong 38 lớp: lớp 0 là nền, lớp 1--37 là giống
        của con vật (12 giống mèo, 25 giống chó).
  \item \st{Nhãn đúng:} ghép mặt nạ trimap chính thức (thú / nền / viền) với nhãn giống của ảnh. Điểm ảnh thuộc thú nhận
        mã giống, điểm nền nhận 0, vùng \emph{viền} (dải mờ quanh biên con vật mà người gán nhãn không chắc chắn) nhận
        giá trị 255 và \st{bị bỏ qua} khi tính loss và metric.
\end{itemize}

\subsection{Câu hỏi nghiên cứu}
\begin{enumerate}
  \item \st{CNN hay Transformer?} Với cùng nguồn pretrained (ImageNet-1k) và số tham số tương đương, U-Net + ResNet-50 hay
        SegFormer-B2 cho mIoU cao hơn trên bài toán 38 lớp, và với chi phí (thời gian, bộ nhớ, tốc độ) ra sao?
  \item \st{Lỗi đến từ đâu?} Phần mIoU bị mất là do vẽ sai biên hay do nhận sai giống? Mô hình nhầm giữa các giống cùng
        loài nhiều hơn nhầm chó $\leftrightarrow$ mèo không?
  \item \st{Mô hình nào cần ít nhãn hơn?} Khi chỉ còn 50\%, 25\%, 10\% ảnh train, khoảng cách giữa hai mô hình thay đổi thế nào?
  \item \st{Các tuyên bố của paper có đứng vững trên tập của nhóm không?} Ví dụ: bộ giải mã toàn MLP gọn nhẹ mà vẫn đủ tốt,
        encoder phân cấp không cần positional encoding (Mục~\ref{sec:paper}).
\end{enumerate}

\section{Tập dữ liệu và EDA}
\subsection{Nguồn, license và cách tải}
Oxford-IIIT Pet được nhóm Visual Geometry Group (Đại học Oxford) và IIIT Hyderabad công bố cùng bài báo CVPR
2012~\cite{parkhi2012}, tại \url{https://www.robots.ox.ac.uk/~vgg/data/pets/}. Tác giả cung cấp dữ liệu
cho mục đích nghiên cứu; việc dùng ảnh phải tôn trọng điều khoản của các trang web gốc nơi ảnh được thu thập (theo file
\code{README} đi kèm bộ nhãn). Nhóm chỉ dùng cho học tập, không phân phối lại ảnh và trích dẫn bài báo gốc.

Mỗi ảnh có nhãn giống, nhãn loài (chó/mèo), hộp bao phần đầu và một mặt nạ \st{trimap} gán mỗi điểm ảnh là thú (1), nền (2)
hoặc viền chưa rõ (3). Hai danh sách chính thức chia dữ liệu thành \code{trainval} (3.680 ảnh) và \code{test} (3.669 ảnh).

Máy chủ của Oxford không truy cập được từ mạng của nhóm, nên \code{data.py} thử máy chủ gốc trước rồi tải hai file
\code{images.tar.gz} và \code{annotations.tar.gz} từ một bản sao trên Hugging Face. Hai file được \st{kiểm tra MD5} trùng
với giá trị thư viện torchvision dùng cho bản gốc, nên dữ liệu giống hệt bản chính thức. Thư mục \code{data/} không được
commit lên GitHub.

\textbf{Nhận xét về số lượng ảnh.}
\begin{itemize}
  \item Tập có 7.349 ảnh của \st{37 giống: 12 giống mèo và 25 giống chó}, mỗi giống khoảng 200 ảnh, chia gần đều giữa

  \item Dữ liệu \st{cân bằng theo giống}: mỗi giống có 80--90 ảnh train, 10 ảnh validation, 88--100 ảnh test. Ít nhất là
        Egyptian Mau và Bombay, do tập gốc có ít ảnh hơn và do bước loại ảnh lỗi bên dưới.
  \item \st{Phát hiện lỗi nhãn:} khi quét trimap, nhóm thấy \st{14 ảnh} (7 trong trainval, 7 trong test) có trimap gán
        \emph{toàn bộ} ảnh là nền, dù ảnh rõ ràng có con vật (ví dụ \code{saint\_bernard\_108}, \code{Egyptian\_Mau\_162},
        \code{Persian\_259}). Giữ lại các ảnh này sẽ dạy mô hình rằng con vật là nền, và trừ điểm mô hình khi nó dự đoán đúng
        trên test. Nhóm \st{loại 14 ảnh này khỏi mọi split} và lưu danh sách trong \code{splits/pet\_split.json}
        (\code{excluded\_empty\_trimap}). Vì vậy split có 3.303 / 370 / 3.662 ảnh thay vì 3.310 / 370 / 3.669.
\end{itemize}

\textbf{Nhận xét về ảnh và mặt nạ.}
\begin{itemize}
  \item \st{Kích thước ảnh không đồng nhất} (Hình~\ref{fig:sizes}): chiều rộng từ 114 đến 3.264 điểm ảnh, chiều cao từ 103
        đến 2.606, trung vị $500\times375$; khoảng 98\% ảnh có cạnh dài không quá 500. Tỉ lệ khung hình đa dạng, cả ngang lẫn dọc.
        Vì vậy cần đưa về một kích thước chung ($384\times384$) để ghép batch. Mọi trimap đều cùng kích thước với ảnh.
  \item 7.346 ảnh là RGB, 3 ảnh có kênh alpha (RGBA); \code{data.py} luôn chuyển về RGB.
  \item \st{Con vật thường chiếm phần nhỏ của ảnh} (Hình~\ref{fig:fractions}): trung vị 27\% điểm ảnh là thú, 60\% là nền và
        11\% là viền. Dải viền khá dày (Hình~\ref{fig:samples}) và bị bỏ qua, nên metric đo chất lượng ở phần thân con vật
        và nền rõ ràng, không đo độ sắc nét ở sát biên. Đây là giới hạn của nhãn gốc mà nhóm sẽ nêu khi thảo luận.
  \item \st{Mất cân bằng mạnh ở mức điểm ảnh} (Hình~\ref{fig:share}): trên tập train, nền chiếm 66\% điểm ảnh có nhãn, mỗi
        giống chỉ 0,4--1,5\%. Mô hình chỉ đoán ``nền'' ở mọi nơi đã đạt pixel accuracy 66\% nhưng mIoU chỉ khoảng
        $66/38\approx1,7\%$. Đây là lý do nhóm chọn mIoU (trung bình đều các lớp) làm metric chính thay vì pixel accuracy.
  \item Trung bình pixel trên 500 ảnh train ngẫu nhiên (R, G, B) là (0,465; 0,430; 0,382), độ lệch chuẩn (0,265; 0,260; 0,266),
        gần thống kê ImageNet, nên dùng chuẩn hoá ImageNet là hợp lý.
  \item Trong một ảnh chỉ có một con vật và một giống, nên mô hình cần nhận ra giống từ toàn cảnh con vật rồi tô nhất quán
        cả vùng. Một lỗi điển hình có thể đoán trước là một con vật bị tô bằng hai giống khác nhau ở hai phần cơ thể.
\end{itemize}

\textbf{Lý do thiết kế.}
\begin{itemize}
  \item \st{Biến đổi hình học áp dụng đồng thời cho ảnh và mặt nạ}, và mặt nạ luôn được nội suy theo điểm gần nhất
        (nearest) để không sinh ra mã lớp không tồn tại ở biên. \code{transforms.v2} của torchvision xử lý việc này khi mặt
        nạ được bọc trong kiểu \code{tv\_tensors.Mask}; Hình~\ref{fig:aug} xác nhận mặt nạ khớp ảnh sau augmentation.
  \item \st{Kích thước $384\times384$:} ảnh gốc có cạnh khoảng 300--500 điểm ảnh (Mục EDA), nên 384 giữ được phần lớn chi
        tiết mà vẫn vừa bộ nhớ GPU T4 với batch 16. Cả hai mô hình nhận đúng cùng đầu vào; metric tính ở độ phân giải này.
  \item \st{Không lật dọc, không xoay mạnh:} ảnh thú cưng hầu như luôn đứng đúng chiều, nên các phép này chỉ tạo ảnh phi thực tế.
  \item \st{Chuẩn hoá ImageNet:} cả hai encoder được huấn luyện trước trên ImageNet-1k với cùng chuẩn hoá này.
\end{itemize}

COCO hay Pascal VOC, vì các tập này có sẵn lớp chó và mèo.

\subsection{Metric}\label{sec:metrics}
Mọi metric được tính trên \st{toàn bộ split} từ một ma trận nhầm lẫn $38\times38$ cộng dồn qua các ảnh (\code{metrics.py}),
bỏ qua điểm ảnh viền. Gọi $TP_c$, $FP_c$, $FN_c$ là số điểm ảnh đúng, dự đoán thừa và bỏ sót của lớp $c$:
\begin{align*}
\mathrm{IoU}_c &= \frac{TP_c}{TP_c + FP_c + FN_c}, &
\mathrm{mIoU} &= \frac{1}{38}\sum_{c=0}^{37}\mathrm{IoU}_c, &
\mathrm{Dice}_c &= \frac{2\,TP_c}{2\,TP_c + FP_c + FN_c}.
\end{align*}
\begin{itemize}
  \item \st{mIoU 38 lớp (metric chính):} mỗi lớp có trọng số như nhau, nên giống hiếm hay khó cũng ảnh hưởng ngang lớp nền.
        Đây là metric chuẩn của bài toán phân đoạn ngữ nghĩa và cũng là metric chính của paper SegFormer.
  \item \st{Mean Dice:} trung bình $\mathrm{Dice}_c$, metric thông dụng thứ hai mà đề bài nêu.
  \item \st{Pixel accuracy:} tỉ lệ điểm ảnh đúng. Chỉ báo cáo để tham khảo, vì nền chiếm phần lớn điểm ảnh (Hình~\ref{fig:share})
        nên metric này dễ cao dù nhận sai giống.
  \item \st{IoU thú/nền:} gộp 37 giống thành một lớp ``thú'' rồi tính IoU. So với mIoU, nó tách riêng chất lượng khoanh hình
        dạng khỏi chất lượng nhận giống.
  \item \st{Chi phí:} số tham số, thời gian huấn luyện, bộ nhớ GPU đỉnh và tốc độ suy luận (ảnh/giây) trên cùng GPU T4.
  \item \st{Độ chính xác giống theo ảnh:} mỗi ảnh test lấy giống được dự đoán nhiều nhất trong các điểm ảnh mô hình cho là
        thú, rồi so với giống thật. Metric này cho biết mô hình có ``hiểu'' đúng con vật không, tách khỏi chất lượng biên.
\end{itemize}

\section{Thảo luận và đối chiếu với paper}\label{sec:discussion}
\subsection{CNN hay Transformer?}
Với cùng dữ liệu, cùng công thức huấn luyện và 15 epoch, SegFormer-B2 đạt mIoU test 89,2\% còn U-Net + ResNet-50 chỉ
46,1\% (Bảng~\ref{tab:main}). Khoảng cách gần như nằm hết ở việc \st{nhận giống}, không phải ở việc khoanh con vật:
\begin{itemize}
  \item IoU thú/nền của hai mô hình gần nhau (98,1\% và 97,0\%). Cả hai đều tách con vật khỏi nền tốt; U-Net chỉ kém
        khoảng 1 điểm.
  \item Độ chính xác giống theo ảnh chênh lớn: 94,7\% so với 72,4\%. Khi U-Net đoán sai giống, cả vùng con vật bị tính
        sai lớp, nên IoU của lớp đúng và của lớp bị đoán nhầm cùng giảm. Đó là lý do mIoU chênh hơn 40 điểm dù hình dạng
        gần như đúng.
\end{itemize}
Đường học (Hình~\ref{fig:curves}) cho thấy hai mô hình học theo hai nhịp rất khác nhau. SegFormer đạt mIoU validation
80\% ngay sau epoch 1 và gần bão hoà từ epoch 3. U-Net chỉ đạt 3,5\% sau epoch 1, rồi tăng chậm và đều tới 48,5\% ở
epoch 15, khi đường cong vẫn còn dốc. Vì vậy \st{kết quả của U-Net là kết quả chưa hội tụ}: với nhiều epoch hơn hoặc
learning rate lớn hơn cho encoder ResNet, khoảng cách chắc chắn sẽ hẹp lại. Nhóm giải thích sự chênh lệch như sau:
\begin{itemize}
  \item \st{Công thức huấn luyện là của SegFormer.} Learning rate $6\times10^{-5}$ cho encoder vốn được paper chọn cho MiT.
        Với ResNet-50 (có BatchNorm, quen với learning rate lớn hơn nhiều), mức này làm encoder gần như đứng yên, nên
        U-Net phải dựa vào bộ giải mã 9 triệu tham số học từ đầu. Đây đúng là hạn chế nhóm đã nêu ở paper
        (Mục~\ref{sec:paper}): so sánh dùng cùng một công thức thì công thức đó có thể thiên vị một kiến trúc.
  \item \st{Nhận giống cần ngữ cảnh toàn cục.} Phân biệt 37 giống dựa vào toàn bộ con vật (mặt, lông, dáng). Self-attention
        của MiT nhìn toàn ảnh ngay từ tầng đầu, còn CNN chỉ có ngữ cảnh rộng ở tầng sâu. Trong U-Net, quyết định ở mỗi điểm
        ảnh còn chịu ảnh hưởng mạnh của skip connection từ tầng nông, nên một con vật có thể bị tô bằng nhiều giống
        (Hình~\ref{fig:examples}).
  \item \st{Bộ giải mã nhẹ vẫn đủ.} Bộ giải mã MLP của SegFormer chỉ 3,2 triệu tham số và dự đoán ở 1/4 độ phân giải, nhưng
        IoU thú/nền vẫn cao hơn U-Net. Trên tập này, nhãn viền bị bỏ qua nên lợi thế ``biên sắc nét'' của U-Net không được
        tính (Mục EDA). Điều này khớp với tuyên bố trung tâm của paper: encoder tốt thì bộ giải mã có thể rất đơn giản.
\end{itemize}

\subsection{Chi phí}
SegFormer-B2 ít tham số hơn (27,4 triệu so với 32,5 triệu) nhưng \st{đắt hơn khi chạy} trên T4 (Bảng~\ref{tab:cost}): mỗi
epoch 128 giây so với 93 giây, bộ nhớ đỉnh 8,3 GB so với 4,6 GB, suy luận 77 so với 134 ảnh/giây. Self-attention ở độ phân
giải 1/4 (dù đã rút gọn key/value) và đặc trưng 768 kênh ở 1/4 độ phân giải của bộ giải mã tốn bộ nhớ hơn tích chập.
Khi đánh giá ở $640\times640$, SegFormer với batch 16 hết bộ nhớ T4 ở lớp trộn của bộ giải mã (cần thêm 2,3 GB), nhóm phải giảm
batch xuống 4. Số tham số nhỏ của paper vì vậy không đồng nghĩa với rẻ hơn trên GPU phổ thông. Tuy vậy, tính theo
\st{chất lượng trên cùng thời gian}, SegFormer vẫn lợi hơn hẳn: sau 1 epoch (khoảng 2 phút) nó đã vượt xa U-Net sau 15 epoch.

\subsection{Lỗi đến từ đâu}
Ma trận gộp nền / mèo / chó (Hình~\ref{fig:species}) và phân tích điểm ảnh thú trên test cho thấy:
\begin{itemize}
  \item U-Net gán đúng giống cho 60,2\% điểm ảnh thú; 30,0\% bị gán \st{sai giống cùng loài}, 8,1\% bị gán \st{sai loài}
        (chủ yếu mèo bị đoán là chó: 15,0\% điểm ảnh mèo), chỉ 1,7\% bị đoán là nền. SegFormer gán đúng 93,8\%; sai
        giống cùng loài 5,4\%, sai loài chỉ 0,1\%, đoán là nền 0,7\%.
  \item Vậy lỗi của cả hai mô hình chủ yếu là \st{nhầm giữa các giống cùng loài}, không phải nhầm chó với mèo hay bỏ sót
        con vật, đúng như dự đoán ở A2.1. U-Net còn có lỗi đặc trưng là tô một con vật bằng nhiều giống (ảnh Bengal\_24 ở
        Hình~\ref{fig:examples}), điều gần như không xảy ra với SegFormer.
  \item Theo từng giống (Hình~\ref{fig:breed}), SegFormer tốt hơn U-Net ở \st{cả 37 giống}; chênh lệch nhỏ nhất vẫn là 18,7 điểm
        (newfoundland). Giống khó nhất với SegFormer là american pit bull terrier (IoU 58,9\%), Ragdoll (62,6\%) và
        staffordshire bull terrier (65,2\%): pit bull và staffordshire gần như giống hệt nhau, còn Ragdoll dễ lẫn với các
        giống mèo lông dài khác (Ragdoll\_62 bị đoán là Maine Coon). Ảnh american\_bulldog\_61 và Ragdoll\_62 là hai trường hợp U-Net đúng còn SegFormer nhầm sang
        giống họ hàng.
  \item \st{Nhãn sai trong tập test.} Nhóm chọn nhóm ảnh ``cả hai cùng sai'' để xem và thấy hai ảnh (Persian\_273,
        Maine\_Coon\_268) có trimap bị \st{đảo ngược}: con vật được gán là nền và nền được gán là thú. Cả hai mô hình đều
        khoanh đúng con vật nhưng bị chấm sai toàn bộ. Đây là loại lỗi nhãn thứ hai, khác với 14 ảnh ``toàn nền'' đã loại ở
        A2.1; nhóm không loại các ảnh này khỏi test sau khi đã thấy kết quả để tránh chỉnh tập test theo mô hình, nhưng
        ghi nhận rằng mIoU báo cáo hơi thấp hơn giá trị thật.
\end{itemize}

% ======================================================================= 8
\subsection{Mô hình nào cần ít nhãn hơn?}
Bảng~\ref{tab:fraction} và Hình~\ref{fig:efficiency} trả lời rõ: \st{SegFormer gần như không phụ thuộc vào lượng nhãn} trong
khoảng thử, còn U-Net sụp nhanh. Khi giảm từ 3.303 xuống 332 ảnh train (10\%), mIoU của SegFormer chỉ giảm từ 89,2\% xuống
83,4\% và độ chính xác giống theo ảnh từ 94,7\% xuống 91,6\%; với chỉ 9 ảnh mỗi giống, nó vẫn tốt hơn U-Net dùng toàn bộ dữ liệu.
U-Net giảm từ 46,1\% xuống 4,8\% mIoU (giống đúng theo ảnh 10,6\%, gần mức đoán bừa trong 37 giống là 2,7\%). Trong khi đó IoU
thú/nền của U-Net vẫn giữ trên 90\% ở mọi mức: tách con vật khỏi nền là việc dễ, cần ít nhãn; còn nhận giống mới là phần cần
dữ liệu, và ở đây encoder MiT pretrained cung cấp đặc trưng ``sẵn sàng'' cho phân biệt giống tốt hơn nhiều. Lưu ý mọi cấu hình
đều chạy 15 epoch, nên cấu hình ít ảnh có ít bước cập nhật hơn; với U-Net (vốn chưa hội tụ), điều này làm khoảng cách lớn thêm.

\subsection{Kiểm chứng hai tuyên bố của paper}
Hai thí nghiệm ở Mục~\ref{sec:robust} dùng 1.000 ảnh test (cứ 3 ảnh lấy 1 theo thứ tự danh sách test, \code{--subset 1000}) để kịp thời gian;
ở $384\times384$ mIoU trên tập con này là 89,0\% (SegFormer) và 43,4\% (U-Net), gần với kết quả trên toàn tập test.

\textbf{Đổi độ phân giải test (Bảng~\ref{tab:resolution}).} Paper cho rằng Mix-FFN thay cho positional encoding giúp mô hình
chịu được độ phân giải test khác lúc train. Trên tập của nhóm, SegFormer giữ mIoU trong khoảng 86,6--89,0\% khi cạnh ảnh đi từ
256 đến 640 (gấp 2,5 lần), tức giảm tối đa 2,4 điểm. Điều này khớp với Bảng 1c của paper (Mix-FFN chỉ giảm 0,7 điểm khi tăng
độ phân giải trên Cityscapes). U-Net thay đổi nhiều hơn theo một hướng đáng chú ý: mIoU \emph{tăng} khi ảnh nhỏ hơn (48,5\% ở 256)
và giảm mạnh khi ảnh lớn hơn (32,3\% ở 640), trong khi IoU thú/nền gần như không đổi (95--97\%). Khi ảnh nhỏ, vùng nhìn của
mỗi điểm ảnh phủ được nhiều phần con vật hơn nên U-Net nhận giống tốt hơn; điều này củng cố giải thích rằng điểm yếu của
U-Net ở đây là thiếu ngữ cảnh toàn cục, không phải thiếu chi tiết.

\textbf{Ảnh hỏng (Bảng~\ref{tab:corruption}, Hình~\ref{fig:robust}).} Paper báo cáo SegFormer-B5 bền hơn hẳn CNN trên
Cityscapes-C. Kết quả của nhóm xác nhận xu hướng này ở cỡ B2 và trên tập khác:
\begin{itemize}
  \item \st{Nhiễu Gauss:} SegFormer gần như không đổi ở $\sigma = 0{,}04$ và $0{,}08$ (88,9\% và 87,5\%), còn 79,5\% ở
        $\sigma = 0{,}16$. U-Net giảm từ 43,4\% xuống 2,6\%: ở mức nặng nhất nó gần như không còn thấy con vật (IoU thú/nền
        5,2\%). Đây là khác biệt lớn nhất, giống với paper (57,8 so với 6,5 trên Cityscapes-C).
  \item \st{Làm mờ và nén JPEG:} cả hai đều giảm mạnh ở mức nặng nhất (SegFormer còn 47,2\% và 45,6\%), nhưng tỉ lệ giữ lại của
        SegFormer vẫn cao hơn: ở mức nặng nhất SegFormer giữ khoảng 53\% và 51\% mIoU ban đầu, U-Net chỉ 28\% và 8\%.
\end{itemize}
Vì U-Net ở đây chưa hội tụ, phép so sánh độ bền giữa hai kiến trúc không hoàn toàn công bằng; kết luận chắc chắn hơn là
\st{SegFormer-B2 tự nó rất bền với nhiễu và đổi độ phân giải}, còn với làm mờ và JPEG nặng thì không mô hình nào bền.

\section{Hạn chế}
\begin{itemize}
  \item \st{U-Net chưa hội tụ.} Đường mIoU validation của U-Net vẫn tăng ở epoch cuối; giới hạn GPU chỉ cho phép 15 epoch và
        một công thức chung. Thí nghiệm công bằng hơn nên tìm learning rate riêng cho U-Net (ví dụ $10^{-4}$ cho encoder) và
        chạy lâu hơn.
  \item \st{Mỗi cấu hình chỉ chạy một lần} (seed 42), nên không có khoảng tin cậy. Với SegFormer, mIoU validation dao động
        khoảng 1--2 điểm giữa các epoch cuối, nên các chênh lệch nhỏ (ví dụ giữa 100\% và 50\% dữ liệu: 89,2 so với 88,2)
        không chắc có ý nghĩa.
  \item \st{Đánh giá thêm trên 1.000 ảnh test} thay vì toàn bộ 3.662 ảnh do thiếu thời gian GPU.
  \item \st{Nhãn:} vùng viền dày bị bỏ qua nên metric không đo được độ sắc nét của biên; tập test còn ảnh có trimap đảo ngược.
  \item \st{Số epoch giữ cố định cho mọi lượng dữ liệu}, nên cấu hình ít ảnh có ít bước cập nhật hơn.
  \item Chưa làm thử chéo trên Pascal VOC và Grad-CAM như dự định mở rộng ở A2.1.
\end{itemize}
